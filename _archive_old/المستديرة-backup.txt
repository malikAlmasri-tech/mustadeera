function renderOwnerStats(bookings){
  const confirmed=bookings.filter(b=>String(b.status||'').toLowerCase()==='confirmed');
  const confirmedFromWebsite=confirmed.filter(b=>!isOwnerManualBooking(b));
  const confirmedManual=confirmed.filter(isOwnerManualBooking);
  const pending=bookings.filter(b=>String(b.status||'').toLowerCase()==='pending');
  const today=getTodayDate();const weekStart=getDateAfter(-6);
  const todayBookings=bookings.filter(b=>String(b.date)===today).length;
  const weekBookings=bookings.filter(b=>isSameOrAfter(b.date,weekStart)&&String(b.date)<=today).length;
  const websiteRevenue=confirmedFromWebsite.reduce((sum,b)=>sum+(Number(b.price)||0),0);
  const manualRevenue=confirmedManual.reduce((sum,b)=>sum+(Number(b.price)||0),0);
  const platformProfit=websiteRevenue*0.10;
  const ownerNetRevenue=websiteRevenue-platformProfit+manualRevenue;
  const rate=bookings.length?Math.round((confirmed.length/bookings.length)*100):0;
  const topField=getTopBy(bookings,b=>String(b.field_id||''),b=>b.field_name||'-');
  const topSource=getTopBy(bookings,b=>String(b.source||'direct').trim()||'direct',b=>String(b.source||'direct').trim()||'direct');
  document.getElementById("ownerTotalBookings").textContent=bookings.length;
  document.getElementById("ownerPendingBookings").textContent=pending.length;
  document.getElementById("ownerConfirmedBookings").textContent=confirmed.length;
  document.getElementById("ownerTodayBookings").textContent=todayBookings;
  document.getElementById("ownerWeekBookings").textContent=`آخر 7 أيام: ${weekBookings}`;
  document.getElementById("ownerConfirmedRevenue").textContent=formatMoney(websiteRevenue);
  document.getElementById("ownerPlatformProfit").textContent=formatMoney(platformProfit);
  document.getElementById("ownerNetRevenue").textContent=formatMoney(ownerNetRevenue);
  document.getElementById("ownerConfirmRate").textContent=rate+'%';
  requestAnimationFrame(()=>{document.getElementById("ownerConfirmRateBar").style.width=Math.min(rate,100)+'%'});
  document.getElementById("ownerTopField").textContent=topField?`${topField.label} (${topField.count})`:'-';
  document.getElementById("ownerTopSource").textContent=topSource?`${topSource.label} (${topSource.count})`:'-';
}

function renderOwnerBookings(){
  if(!ownerData)return;
  let bookings=ownerData.bookings||[];
  const date=document.getElementById("ownerDateFilter").value;
  const fieldId=document.getElementById("ownerFieldFilter").value;
  if(!showAllOwnerBookings)bookings=getVisibleBookings(bookings,30,false);

  if(!bookings.length){el.innerHTML='<div class="card" style="text-align:center;color:var(--soft)">لا توجد حجوزات مطابقة</div>';return}
  const{active,finished}=splitFinishedBookings(bookings);
  function renderGrouped(list){
    const grouped={};list.forEach(b=>{const d=String(b.date);if(!grouped[d])grouped[d]=[];grouped[d].push(b)});
    const sortedDays=Object.keys(grouped).sort();const today=getTodayDate();const tomorrow=getDateAfter(1);
    function getDayLabel(dateStr){if(dateStr===today)return `اليوم — ${getArabicDayName(dateStr)} ${getShortDate(dateStr)}`;if(dateStr===tomorrow)return `غداً — ${getArabicDayName(dateStr)} ${getShortDate(dateStr)}`;return `${getArabicDayName(dateStr)} ${getShortDate(dateStr)}`}
    return sortedDays.map(date=>{
      const dayBookings=grouped[date].sort((a,b)=>Number(a.hour)-Number(b.hour));
      const confirmed=dayBookings.filter(b=>b.status==='confirmed').length;const pending=dayBookings.filter(b=>b.status==='pending').length;const cancelled=dayBookings.filter(b=>b.status==='cancelled'||b.status==='rejected').length;
      const isToday=date===today;const isPast=date<today;
      return `<div style="margin-bottom:22px">
        <div class="day-group-header" style="display:flex;justify-content:space-between;align-items:center;background:${isToday?'linear-gradient(140deg,var(--forest),var(--emerald))':isPast?'var(--surface-muted)':'var(--soft-green)'};color:${isToday?'#fff':isPast?'var(--soft)':'var(--forest)'};border-radius:14px;padding:12px 15px;margin-bottom:11px;font-size:13px;font-weight:800;box-shadow:${isToday?'var(--shadow-green)':'var(--shadow-xs)'};letter-spacing:-.01em">
          <span>📅 ${getDayLabel(date)}</span>
          <span style="display:flex;gap:8px;font-size:11px;font-weight:800;opacity:.9">
            ${confirmed>0?`<span style="background:rgba(21,128,61,.18);padding:2px 9px;border-radius:999px">✓ ${confirmed}</span>`:''}
            ${pending>0?`<span style="background:rgba(245,158,11,.18);padding:2px 9px;border-radius:999px">⏳ ${pending}</span>`:''}
            ${cancelled>0?`<span style="background:rgba(220,38,38,.12);padding:2px 9px;border-radius:999px">✕ ${cancelled}</span>`:''}
          </span>
        </div>
        ${dayBookings.map(b=>renderSingleBookingCard(b)).join('')}
      </div>`
    }).join('')
  }
  el.innerHTML=sectionTitle('الحجوزات الحالية والقادمة',active.length)+(active.length?renderGrouped(active):'<div class="card" style="text-align:center;color:var(--soft);margin-bottom:11px">لا توجد حجوزات حالية أو قادمة</div>')+sectionTitle('الحجوزات المنتهية',finished.length)+renderGrouped(finished)
}
